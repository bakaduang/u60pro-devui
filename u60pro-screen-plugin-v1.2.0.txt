  const buildTar = (files) => {
    const enc = new TextEncoder()
    const chunks = []
    const header = (name, size, mode, type) => {
      const h = new Uint8Array(512)
      const put = (s, off, len) => h.set(enc.encode(s).subarray(0, len), off)
      const oct = (n, len) => n.toString(8).padStart(len - 1, '0') + '\0'
      put(name, 0, 100); put(oct(mode, 8), 100, 8); put(oct(0, 8), 108, 8); put(oct(0, 8), 116, 8)
      put(oct(size, 12), 124, 12); put(oct(Math.floor(Date.now() / 1000), 12), 136, 12)
      h.fill(32, 148, 156); put(type, 156, 1); put('ustar\0', 257, 6); put('00', 263, 2)
      let sum = 0
      for (const b of h) sum += b
      put(sum.toString(8).padStart(6, '0') + '\0 ', 148, 8)
      return h
    }
    const dirs = new Set()
    Object.keys(files).forEach((p) => { const i = p.lastIndexOf('/'); if (i > 0) dirs.add(p.slice(0, i + 1)) })
    dirs.forEach((d) => chunks.push(header(d, 0, 0o755, '5')))
    Object.keys(files).forEach((p) => {
      const data = enc.encode(files[p])
      if (enc.encode(p).length >= 100) throw new Error(`文件名过长：${p}`)
      chunks.push(header(p, data.length, p.endsWith('.sh') ? 0o755 : 0o644, '0'), data)
      const pad = (512 - (data.length % 512)) % 512
      if (pad) chunks.push(new Uint8Array(pad))
    })
    chunks.push(new Uint8Array(1024))
    return new Blob(chunks, { type: 'application/x-tar' })
  }

        if (buf[i] !== marker[0]) continue
        let j = 1
        while (j < marker.length && buf[i + j] === marker[j]) j++
        if (j === marker.length) return true

      return false

      'echo; echo "== 系统日志 =="; logread 2>/dev/null | grep -i -e datad -e segfault | tail -n 15; dmesg 2>/dev/null | grep -i -e datad -e segfault -e "killed process" | tail -n 8',
      `echo; echo "== 文件 =="; ls -la ${DIR} ${UI} ${UI}/functions 2>/dev/null`,
    ].join('\n'), 20000)
    const id = `${ID}_diag`
    document.getElementById(id)?.remove()
    const mask = document.createElement('div')
    mask.className = 'mask'; mask.id = id
    mask.style.cssText = `${MASK_POS};z-index:2001`
    const box = document.createElement('div')
    box.className = 'modal'
    box.style.cssText = 'width:94%;max-width:640px;max-height:86vh;display:flex;flex-direction:column;gap:10px;padding:14px'
    const ta = document.createElement('textarea')
    ta.readOnly = true
    ta.style.cssText = 'width:100%;height:60vh;box-sizing:border-box;font:11px/1.45 monospace;white-space:pre;overflow:auto;background:rgba(0,0,0,.25);color:inherit;border-radius:8px;padding:8px'
    ta.value = `U60 屏幕管理 ${PLUGIN_VERSION} · UI ${UI_VERSION}\n${out}`
    const row = document.createElement('div'); row.style.cssText = 'display:flex;gap:8px'
    const cp = document.createElement('button'); cp.className = 'btn'; cp.textContent = '复制'; cp.style.flex = '1'
    const cl = document.createElement('button'); cl.className = 'btn'; cl.textContent = '关闭'; cl.style.flex = '1'
    cp.onclick = async () => {
      try { await navigator.clipboard.writeText(ta.value); toast('已复制', 'green') }
      catch (e) { ta.focus(); ta.select(); try { document.execCommand('copy'); toast('已复制', 'green') } catch (e2) { toast('请长按手动复制', 'pink') } }
    }
    cl.onclick = () => mask.remove()
    row.append(cp, cl); box.append(ta, row); mask.appendChild(box); document.body.appendChild(mask)
  }

  const doLegacy = async () => {
    const s = await refresh()
    const builtin = String(s.DATAD_EXE || '').split(/\s+/).filter(Boolean).some((p) => p !== LEGACY_DATAD)
    const items = []
    if (Number(s.RC_DEVUI) > 0) items.push('删除 /etc/rc.local 中旧版屏幕自启行（改由 ufi_tools_boot.sh 启动）')
    if (Number(s.RC_DATAD) > 0 && builtin) items.push('删除 /etc/rc.local 中旧版 datad 自启行（已检测到内置 datad）')
    const legacyIdle = s.LEGACY_DATAD === '1' && builtin && !String(s.DATAD_EXE || '').includes(LEGACY_DATAD)
    if (legacyIdle) items.push('删除未在使用的旧版 datad 目录 /data/plugins/zwrt-datad')
    if (!items.length) {
      log(Number(s.RC_DATAD) > 0 && !builtin ? '旧版 datad 自启仍在使用（未检测到内置 datad），已保留' : '没有需要清理的旧版残留')
      return
    }
    if (!await askConfirm('清理旧版残留', items, '清理')) return
    const cmd = [RC_DEVUI_CLEAN]
    if (Number(s.RC_DATAD) > 0 && builtin) cmd.push("sed -i '/u60pro_devui_datad/d' /etc/rc.local")
    if (legacyIdle) cmd.push('rm -rf /data/plugins/zwrt-datad')
    cmd.push('echo ok')
    await shell(cmd.join('\n'), 10000)
    log('旧版残留已清理')
  }

  const doUninstall = async () => {
    const ok = await askConfirm('卸载屏幕管理', [
      '1. 停止社区屏幕程序并恢复原厂屏幕；',
      '2. 删除开机自启行（ufi_tools_boot.sh 与 rc.local 中的屏幕自启）；',
      `3. 删除 ${DIR}（含渲染程序、页面、锁屏密码、你放进 ui/functions 的自定义页）；`,
      '4. 不会触碰 datad、小海关或其他插件。',
      '插件本身请在插件商店中移除。',
    ], '确认卸载', true)
    if (!ok) return
    await restoreOem()
    await setBoot(false)
    await shell(`rm -rf ${DIR} /tmp/devui-x /tmp/u60pro-devui.log /tmp/u60pro-devui-boot.log; echo ok`, 15000)
    log('已卸载并恢复原厂屏幕')
    toast('已卸载，原厂屏幕已恢复', 'green')
  }

  const actions = {
    install: lock(doInstall),
    ui: lock(async () => {
      if (!await askConfirm('重装页面 UI', [`安装内置页面 ${UI_VERSION}，保留锁屏密码和你自己的自定义页面。`], '重装')) return
      await installUi(log)
      toast('页面 UI 已更新，屏幕约 1 秒后刷新', 'green')
    }),
    start: lock(async () => {
      if (state.BIN !== '1') throw new Error('请先安装渲染程序')
      await writeStartScript(); await startRenderer(); log('已启动')
    }),
    restart: lock(async () => {
      if (state.BIN !== '1') throw new Error('请先安装渲染程序')
      await writeStartScript(); await startRenderer(); log('已重启')
    }),
    stop: lock(async () => { await restoreOem(); log('已停止，原厂屏幕已恢复（开机自启设置未改变）') }),
    boot: lock(async () => {
      const on = state.BOOT !== '1'
      if (on && state.BIN !== '1') throw new Error('请先安装渲染程序')
      if (on) await writeStartScript()
      await setBoot(on)
      log(on ? '已开启开机自启' : '已关闭开机自启')
    }),
    refresh: lock(async () => { log('已刷新') }),
    check: lock(async () => {
      saveGh()
      const rel = await fetchRelease(log)
      showRelease(rel)
      log(`GitHub 最新版本 ${rel.tag}`)
    }),
    diag: lock(doDiag),
    legacy: lock(doLegacy),
    uninstall: lock(doUninstall),
    preset: () => {
      $(`#${ID}_url`).value = PRESETS.remix.url
      $(`#${ID}_sha`).value = PRESETS.remix.sha
    },
    close: () => { if (typeof closeModal === 'function') closeModal(`#${MODAL}`); else root.style.display = 'none' },
  }

  const open = () => {
    if (!root) {
      document.getElementById(MODAL)?.remove()
      const wrap = document.createElement('div')
      wrap.innerHTML = html.trim()
      root = wrap.firstElementChild
      document.body.appendChild(root)
      root.addEventListener('click', (e) => {
        const t = e.target.closest('[data-act]')
        if (t && actions[t.dataset.act]) { e.preventDefault(); actions[t.dataset.act]() }
      })
      root.querySelectorAll(`input[name="${ID}_src"]`).forEach((r) => { r.addEventListener('change', syncSrc) })
      const saved = lsGet(LS_SRC) || 'online'
      const radio = root.querySelector(`input[name="${ID}_src"][value="${saved}"]`) || root.querySelector(`input[name="${ID}_src"][value="online"]`)
      radio.checked = true
      $(`#${ID}_gh`).value = ghProxy()
      $(`#${ID}_url`).value = lsGet(LS_URL)
      $(`#${ID}_sha`).value = lsGet(LS_SHA)
      $(`#${ID}_file`).addEventListener('change', (e) => { inspectFile(e.target.files && e.target.files[0]).catch((err) => setText('fileinfo', String(err))) })
      syncSrc()
    }
    if (typeof showModal === 'function') showModal(`#${MODAL}`)
    else root.style.display = 'flex'
    refresh().catch((e) => { log(`读取状态失败：${e.message || e}`) })
  }

  const waitFor = async (selector, timeout = 10000) => {
    const until = Date.now() + timeout
    let node = document.querySelector(selector)
    while (!node && Date.now() < until) { await sleep(150); node = document.querySelector(selector) }
    return node
  }

  const mount = async () => {
    const box = await waitFor('.functions-container .collapse_box', 30000)
    if (!box) { console.warn(`[${ID}] 功能列表未就绪，未添加入口`); return }
    document.getElementById(`${ID}_btn`)?.remove()
    const btn = document.createElement('button')
    btn.id = `${ID}_btn`
    btn.className = 'btn'
    btn.type = 'button'
    btn.textContent = '屏幕管理'
    btn.addEventListener('click', open)
    box.appendChild(btn)
  }

  mount().catch((e) => console.error(`[${ID}] 初始化失败`, e))
})()
</script>
